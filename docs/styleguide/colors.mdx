export const Swatch = ({ name, from }) => (
  <div style={{ display: "flex", alignItems: "center", gap: "0.75rem", padding: "0.3rem 0", minWidth: 0 }}>
    <span
      aria-hidden="true"
      style={{
        width: "2.25rem",
        height: "2.25rem",
        flexShrink: 0,
        borderRadius: "6px",
        background: `var(${name})`,
        border: "1px solid var(--ifm-color-emphasis-300)",
      }}
    />
    <div style={{ minWidth: 0 }}>
      <code style={{ fontSize: "0.75rem", wordBreak: "break-all" }}>{name}</code>
      {from && (
        <div style={{ fontSize: "0.7rem", opacity: 0.6, marginTop: "0.15rem", wordBreak: "break-all" }}>
          ↳ {from}
        </div>
      )}
    </div>
  </div>
);

export const Swatches = ({ names }) => (
  <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fill, minmax(240px, 1fr))", gap: "0.25rem 1.5rem", marginBottom: "1.5rem" }}>
    {names.map((n) => {
      const [name, from] = Array.isArray(n) ? n : [n, null];
      return <Swatch key={name} name={name} from={from} />;
    })}
  </div>
);

Every custom property from `src/css/variables.scss`, resolved live — so toggling
the theme switch shows both palettes. Swatch styling is inline on purpose, to
keep this page from adding anything to the production stylesheet.

Where an Infima token is an alias onto a brand variable, the target is noted
beneath it with a `↳`. Some resolve differently per theme, which is written as
`light / dark`.

## Emphasis scale

The borders-and-fills ramp, aliased onto the gray ramp above. It's **inverted**
in dark mode — `emphasis-0` is `gray-0` in light and `gray-1000` in dark — so
toggling the theme reverses this row. That inversion is why these carry no
fixed `↳` target. `emphasis-200` is the one the partials lean on most.

| --- | --- | --- |
| `--root-wrapper-width` | `110rem` | Max width of navbar, footer, and main wrapper |
| `--doc-sidebar-width` | `300px` | Doc sidebar width |
| `--h1-base-font-size` | `2.25rem` | Shared `h1` size token |
| `--ifm-h4-font-size` | `1.313rem` | Heading level 4 |
| `--ifm-font-size-base` | `1rem` | Body copy |
| `--ifm-font-weight-bold` | `800` | Bold text |
| `--ifm-leading-desktop` | `1.75` | Line height multiplier |
| `--ifm-navbar-height` | `80px` | Navbar height |
| `--ifm-global-radius` | `0rem` | Global corner radius |
| `--ifm-code-border-radius` | `0.375rem` | Inline code corners |
| `--ifm-alert-border-left-width` | `2px` | Admonition left rule |
| `--ifm-container-width-xl` | `1400px` | Wide container breakpoint |
